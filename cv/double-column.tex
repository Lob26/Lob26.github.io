% Section heading: rule UNDERNEATH, never on the same baseline (OCR merges it
% into the word -- see ivy.tex).
\newcommand{\cvsection}[1]{%
  \needspace{3\baselineskip}%
  \par\vspace{2.2ex}%
  {\color{ink}\large\bfseries\MakeUppercase{#1}}\par
  \vspace{0.4ex}%
  {\color{navy}\rule{\linewidth}{0.9pt}}\par
  \vspace{0.9ex}}

% Skill pill. Kept as plain text inside a drawn box: the glyphs stay in the
% text stream, so pills cost nothing at extraction time.
\newcommand{\pill}[1]{%
  \tikz[baseline=(p.base)]{%
    \node[draw=pillline,line width=0.4pt,rounded corners=2.5pt,
          inner xsep=5pt,inner ysep=3.5pt,anchor=base,
          text=slate,font=\footnotesize] (p) {#1};}%
  \hspace{1pt}}
