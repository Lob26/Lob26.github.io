% --- Section headings: rule under the text, not beside it -------------------
% moderncv classic puts a \rule{\hintscolumnwidth}{0.95ex} in the LEFT cell of
% the same row as the heading, so a heavy black bar sits on the heading's own
% baseline. Rasterise the page and OCR merges bar and word: "Summary" came back
% as "mes SUM alry", "Skills" and "Projects" were lost outright. Section names
% are exactly the tokens an ATS uses to segment a CV, so losing them is the
% worst possible failure. Same rule, moved under the heading, where OCR reads
% it as an ordinary separator. Redefinition mirrors moderncvbodyi.sty.
\makeatletter
\RenewDocumentCommand{\section}{sm}{%
  \par\addvspace{2.5ex}%
  \phantomsection{}% reset the anchor for hyperrefs
  \addcontentsline{toc}{section}{#2}%
  \cvitem[0ex]{}{%
    \strut\sectionstyle{#2}%
    \newline\vspace*{-1.2ex}%
    {\color{bodyrulecolor}\rule{\maincolumnwidth}{0.9pt}}}%
  \par\nobreak\addvspace{1ex}\@afterheading}
\makeatother

% --- Custom helpers ---------------------------------------------------------
% Body description under each bullet: slightly warmer than before.
\newcommand{\impact}[1]{{\color{slate}\small #1}}
% Continuation paragraph inside the same bullet. Each impact statement used to
% be one 60-110 word block; a human skims the CV after the ATS passes it, and a
% wall of text gets skipped even when the numbers inside it are good. Splitting
% each bullet into "what I did" and "what it produced" costs no words and makes
% the metrics land on their own line.
\newcommand{\impactnext}[1]{\\[0.4em]\impact{#1}}
% Metric: navy bold + a soft whisper background so the number pops.
\newcommand{\metric}[1]{%
  \begingroup
  \setlength{\fboxsep}{2pt}%
  \colorbox{whisper}{\color{navy}\bfseries\strut #1}%
  \endgroup
}
% Stakeholder chip: bracketed, sky italic, slightly stronger than before.
\newcommand{\stakeholders}[1]{{\color{azure}\footnotesize\textsf{\textit{[#1]}}}}
% Role label inside bullets (the bold lead-in): navy bold.
\newcommand{\rolebold}[1]{{\color{navy}\bfseries #1}}
% Callout box for Profile and project human/technical pairs.
\newcommand{\callout}[1]{%
  \begingroup
  \setlength{\fboxsep}{0.8em}%
  \noindent\fcolorbox{sky!40}{whisper}{%
    \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}%
      \color{slate}\small #1%
    \end{minipage}%
  }%
  \endgroup
}
% Visual separator: a softer, two-tone rule (navy dot + grey line).
\newcommand{\softrule}{%
  \vspace{0.45em}
  \noindent\hspace{0pt}{\color{navy}\rule{6pt}{2pt}}{\color{softgray!40}\rule[0.5pt]{0.92\linewidth}{0.4pt}}\hspace{0pt}%
  \vspace{0.35em}
}
